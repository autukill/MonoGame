cmake_minimum_required(VERSION 3.24)
project(Dx12RootSignatureProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(Dx12RootSignatureProbe Main.cpp)
target_compile_features(Dx12RootSignatureProbe PRIVATE cxx_std_17)
target_compile_definitions(Dx12RootSignatureProbe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(Dx12RootSignatureProbe PRIVATE d3d12 dxgi d3dcompiler)
