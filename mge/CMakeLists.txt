cmake_minimum_required(VERSION 3.24)
project(MyGameEngineMonoGameDx12 LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 only")
endif()
get_filename_component(MGE_ROOT "${CMAKE_CURRENT_LIST_DIR}/.." ABSOLUTE)
set(MGE_NATIVE "${MGE_ROOT}/native/monogame")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${MGE_NATIVE}/external/sdl2/sdl" sdl)
file(GLOB native_sources CONFIGURE_DEPENDS "${MGE_NATIVE}/common/*.cpp" "${MGE_NATIVE}/directx12/*.cpp" "${MGE_NATIVE}/sdl/*.cpp" "${MGE_NATIVE}/xaudio/*.cpp")
option(MGE_DX12_STARTUP_DIAGNOSTICS "Build an isolated DX12 failure diagnostic runtime" OFF)
function(configure_native target)
 target_compile_features(${target} PRIVATE cxx_std_17)
 target_compile_definitions(${target} PRIVATE DLL_EXPORT MG_SDL2 MG_DIRECTX12 MG_XAUDIO)
 if(MGE_DX12_STARTUP_DIAGNOSTICS)
  target_compile_definitions(${target} PRIVATE MGE_DX12_STARTUP_DIAGNOSTICS)
 endif()
 target_include_directories(${target} PRIVATE "${MGE_NATIVE}/include" "${MGE_NATIVE}/directx12" "${MGE_ROOT}/external/stb" "${MGE_EFFECTS}")
 target_link_libraries(${target} PRIVATE SDL2-static dxguid dxgi d3d12 winmm imm32 user32 gdi32 advapi32 setupapi ole32 oleaut32 version shell32)
endfunction()
add_library(mgruntime SHARED ${native_sources})
configure_native(mgruntime)
option(MGE_LIFECYCLE "Build separate native lifecycle executable" ON)
if(MGE_LIFECYCLE)
 add_executable(dx12-lifecycle ${native_sources} "${CMAKE_CURRENT_LIST_DIR}/tests/native/NativeLifecycleProbe.cpp")
 configure_native(dx12-lifecycle)
 target_compile_definitions(dx12-lifecycle PRIVATE MGE_BUFFER_FIRST_WRITE_PROBE)
endif()
if(MGE_DX12_STARTUP_DIAGNOSTICS)
 add_executable(dx12-diagnostics "${MGE_NATIVE}/directx12/StartupDiagnostics.cpp" "${CMAKE_CURRENT_LIST_DIR}/tests/native/StartupDiagnosticsProbe.cpp")
 configure_native(dx12-diagnostics)
endif()
