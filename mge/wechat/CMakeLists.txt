cmake_minimum_required(VERSION 3.26)
project(MonoGameWeChat LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
 message(FATAL_ERROR "The WeChat backend requires Emscripten 3.1.56")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(NATIVE "${ROOT}/native/monogame")
set(PORT "${CMAKE_CURRENT_LIST_DIR}/native")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL2_DISABLE_SDL2MAIN ON CACHE BOOL "" FORCE)
foreach(feature RENDER HAPTIC HIDAPI POWER LOADSO FILESYSTEM SENSOR LOCALE MISC OPENGL DISKAUDIO DUMMYAUDIO DUMMYVIDEO OFFSCREEN VIRTUAL_JOYSTICK AUDIO)
 set(SDL_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(SDL_VIDEO ON CACHE BOOL "" FORCE)
set(SDL_EVENTS ON CACHE BOOL "" FORCE)
set(SDL_JOYSTICK ON CACHE BOOL "" FORCE)
set(SDL_FILE ON CACHE BOOL "" FORCE)
set(SDL_OPENGLES ON CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/.sources/SDL" sdl2)
set_target_properties(SDL2-static PROPERTIES DEBUG_POSTFIX "" OUTPUT_NAME SDL2 ARCHIVE_OUTPUT_DIRECTORY "${MGE_OUTPUT}")
add_library(mgruntime STATIC
 "${NATIVE}/common/MGI.cpp" "${NATIVE}/common/MG_Asset.cpp" "${NATIVE}/common/mg_audio.cpp" "${NATIVE}/common/mg_hash.cpp" "${NATIVE}/common/mg_library.cpp"
 "${PORT}/opengl/OpenGLContext.cpp" "${PORT}/opengl/MGG_OpenGL.cpp" "${PORT}/sdl/MGP_sdl.cpp" "${PORT}/web/MGP_web.cpp"
 "${PORT}/audio/MGA_wechat.cpp" "${PORT}/audio/MGM_wechat.cpp")
target_compile_definitions(mgruntime PRIVATE MG_SDL2 MG_OPENGL MG_WECHAT)
target_compile_options(mgruntime PRIVATE -fexceptions -Wno-warn-absolute-paths)
target_include_directories(mgruntime PRIVATE "${PORT}/include" "${NATIVE}/include" "${NATIVE}/common" "${PORT}/sdl" "${PORT}/opengl" "${CMAKE_CURRENT_LIST_DIR}/.sources/SDL/include" "${ROOT}/external/stb" "${MGE_EFFECTS}")
target_link_libraries(mgruntime PRIVATE SDL2-static)
set_target_properties(mgruntime PROPERTIES PREFIX "" OUTPUT_NAME mgruntime ARCHIVE_OUTPUT_DIRECTORY "${MGE_OUTPUT}")
